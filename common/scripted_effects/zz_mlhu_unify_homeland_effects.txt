# Unify Your Home Lands (MLHU) - effects.

# Old Imperator bug: claims added by script are not picked up by the Declare War screen
# ("War without Casus Belli") until the country's war-goal data is rebuilt. In the UI the
# known fix is to queue a building and cancel it; this does the same from script by adding
# one level of a building the capital does not have and removing it again (net change: none).
# Only buildings with 0 levels are used, so an existing building can never be lost.
# Country scope.
mlhu_refresh_claims_effect = {
	capital_scope = {
		if = {
			limit = { has_city_status = yes }
			if = {
				limit = { num_of_library_building = 0 }
				add_building_level = library_building
				remove_building_level = library_building
			}
			else_if = {
				limit = { num_of_court_building = 0 }
				add_building_level = court_building
				remove_building_level = court_building
			}
			else_if = {
				limit = { num_of_theathre_building = 0 }
				add_building_level = theathre_building
				remove_building_level = theathre_building
			}
			else_if = {
				limit = { num_of_aqueduct_building = 0 }
				add_building_level = aqueduct_building
				remove_building_level = aqueduct_building
			}
		}
		else = {
			if = {
				limit = { num_of_basic_settlement_infratructure_building = 0 }
				add_building_level = basic_settlement_infratructure_building
				remove_building_level = basic_settlement_infratructure_building
			}
			else_if = {
				limit = { num_of_latifundia_building = 0 }
				add_building_level = latifundia_building
				remove_building_level = latifundia_building
			}
			else_if = {
				limit = { num_of_slave_mine_building = 0 }
				add_building_level = slave_mine_building
				remove_building_level = slave_mine_building
			}
			else_if = {
				limit = { num_of_hill_fort = 0 }
				add_building_level = hill_fort
				remove_building_level = hill_fort
			}
		}
	}
}

# Measures how much of our homeland we own (variables mlhu_homeland_total, mlhu_homeland_owned
# and mlhu_owned_pct, 0-100). Run when the mission starts and every month by event mlhu.1.
# Country scope.
mlhu_update_progress_effect = {
	set_variable = { name = mlhu_homeland_total value = 0 }
	set_variable = { name = mlhu_homeland_owned value = 0 }
	save_scope_as = mlhu_country
	every_in_list = {
		variable = mlhu_homeland
		scope:mlhu_country = {
			change_variable = { name = mlhu_homeland_total add = 1 }
		}
		if = {
			limit = { owner = scope:mlhu_country }
			scope:mlhu_country = {
				change_variable = { name = mlhu_homeland_owned add = 1 }
			}
		}
	}
	set_variable = { name = mlhu_owned_pct value = 0 }
	if = {
		limit = { var:mlhu_homeland_total > 0 }
		set_variable = { name = mlhu_owned_pct value = var:mlhu_homeland_owned }
		change_variable = { name = mlhu_owned_pct multiply = 100 }
		change_variable = { name = mlhu_owned_pct divide = var:mlhu_homeland_total }
	}
}

# Clears the mission's homeland list and progress variables (mission ended or aborted).
mlhu_clear_progress_effect = {
	clear_variable_list = mlhu_homeland
	remove_variable = mlhu_homeland_total
	remove_variable = mlhu_homeland_owned
	remove_variable = mlhu_owned_pct
}

# Claims on every homeland territory we don't own, made visible to the Declare War screen.
mlhu_claim_homeland_effect = {
	custom_tooltip = mlhu_task_1_tt
	hidden_effect = {
		save_scope_as = mlhu_country
		every_in_list = {
			variable = mlhu_homeland
			limit = { NOT = { owner = scope:mlhu_country } }
			add_claim = scope:mlhu_country
		}
		mlhu_refresh_claims_effect = yes
	}
}

# One tree or the other (v1.13): remembers whether an AI nation has started an Invictus
# mission tree, so the mission's potential reads one variable instead of ~500 task checks.
# Called twice a year by mlhu_tree_cache_pulse. Country scope.
mlhu_refresh_tree_cache_effect = {
	if = {
		limit = { mlhu_invictus_tree_started_trigger = yes }
		set_variable = mlhu_invictus_tree_started
	}
	else = {
		set_variable = {
			name = mlhu_tree_check_cd
			days = 180
		}
	}
}
