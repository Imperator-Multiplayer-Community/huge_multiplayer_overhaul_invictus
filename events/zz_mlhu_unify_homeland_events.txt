# Unify Your Home Lands (MLHU) - events.
# mlhu.1: monthly, hidden - measures how much of the homeland we own (the tasks read the result).
# mlhu.11-17: one event per task, giving its reward. Rewards are given here and not inside the
# mission tasks, because effects and list triggers inside mission tasks are unreliable in Imperator.
namespace = mlhu

mlhu.1 = {
	type = country_event
	hidden = yes

	trigger = {
		mlhu_mission_running_trigger = yes
	}

	immediate = {
		mlhu_update_progress_effect = yes
	}
}

# 1. Call of Kinship
mlhu.11 = {
	type = country_event
	title = mlhu.11.t
	desc = mlhu.11.desc
	picture = council_country

	option = {
		name = mlhu.11.a
		mlhu_claim_homeland_effect = yes
	}
}

# 2. Secure the Heartland
mlhu.12 = {
	type = country_event
	title = mlhu.12.t
	desc = mlhu.12.desc
	picture = greeks_marching

	option = {
		name = mlhu.12.a
		add_stability = 5
		add_country_modifier = {
			name = mlhu_rallied_kin
			duration = 3650
		}
	}
}
