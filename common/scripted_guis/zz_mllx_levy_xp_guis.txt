# Levy XP (MLLX) - hooked into gui/military_view.gui raise / disband buttons.
mllx_single_levy_xp_set_start_value = {
	scope = country
	effect = {
		mllx_set_levy_starting_xp_effect = yes
	}
}

mllx_all_levy_xp_set_start_value = {
	scope = country
	effect = {
		mllx_set_levy_starting_xp_effect = yes
	}
}

# Disbanding (v1.10): the game engine decides which armies are levies and disbands
# them (Disband All Levies / per-governorship disband). Our script no longer
# destroys anything - it used to destroy EVERY non-legion army, including armies
# granted by missions and events (e.g. Macrobian Mahout's war elephants).
# The buttons run: mllx_levy_xp_before_disband -> engine disband -> mllx_levy_xp_after_disband.
# "before" works out the military experience each tagged levy would pay; "after"
# takes back the share of every army that is still alive, so only the armies the
# engine really disbanded pay out.

mllx_levy_xp_before_disband = {
	scope = country

	ai_is_valid = {
		always = no
	}

	effect = {
		set_variable = {
			name = mllx_pending_levy_xp
			value = 0
		}
		every_army = {
			limit = {
				has_legion_trigger = no
				has_variable = levy_start_xp
				exists = commander
				commander = {
					is_mercenary = no
					OR = {
						is_governor = yes
						is_ruler = yes
					}
				}
			}
			save_scope_as = mllx_levy_unit
			set_variable = {
				name = mllx_pending_xp
				value = experience_percentage
			}
			change_variable = {
				name = mllx_pending_xp
				multiply = 100
			}
			change_variable = {
				name = mllx_pending_xp
				subtract = scope:mllx_levy_unit.var:levy_start_xp
			}
			change_variable = {
				name = mllx_pending_xp
				multiply = 0.075
			}
			if = {
				limit = { var:mllx_pending_xp < 0 }
				set_variable = {
					name = mllx_pending_xp
					value = 0
				}
			}
			if = {
				limit = { var:mllx_pending_xp > 10 }
				set_variable = {
					name = mllx_pending_xp
					value = 10
				}
			}
			ROOT = {
				change_variable = {
					name = mllx_pending_levy_xp
					add = scope:mllx_levy_unit.var:mllx_pending_xp
				}
			}
		}
	}
}

mllx_levy_xp_after_disband = {
	scope = country

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { has_variable = mllx_pending_levy_xp }
			# armies still alive were not disbanded: they pay nothing
			every_army = {
				limit = { has_variable = mllx_pending_xp }
				save_scope_as = mllx_levy_unit
				ROOT = {
					change_variable = {
						name = mllx_pending_levy_xp
						subtract = scope:mllx_levy_unit.var:mllx_pending_xp
					}
				}
				remove_variable = mllx_pending_xp
			}
			if = {
				limit = { var:mllx_pending_levy_xp > 0 }
				add_military_experience = {
					value = var:mllx_pending_levy_xp
				}
			}
			remove_variable = mllx_pending_levy_xp
		}
	}
}
