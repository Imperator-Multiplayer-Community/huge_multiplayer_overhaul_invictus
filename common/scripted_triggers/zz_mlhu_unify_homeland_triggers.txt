# Unify Your Home Lands (MLHU) - who can start the mission.
# Only the region of our capital counts as our home lands.

# The culture that is the majority in our capital is the majority in a territory of our
# capital's region that we don't own.
mlhu_capital_kin_nearby_trigger = {
	capital_scope = {
		exists = region
		region = {
			any_region_province = {
				is_sea = no
				has_owner = yes
				NOT = { owner = root }
				dominant_province_culture = root.capital_scope.culture
			}
		}
	}
}

# Same check for our primary culture (fallback when the people of our capital have nothing
# left to unite in the region).
mlhu_primary_kin_nearby_trigger = {
	capital_scope = {
		exists = region
		region = {
			any_region_province = {
				is_sea = no
				has_owner = yes
				NOT = { owner = root }
				dominant_province_culture = root.culture
			}
		}
	}
}

# We own at least $PCT$% of our homeland. The share is measured once a month by event mlhu.1
# (and when the mission starts), not by a list trigger inside the task, because list triggers
# in mission tasks are unreliable in Imperator.
mlhu_owns_homeland_share_trigger = {
	has_variable = mlhu_owned_pct
	var:mlhu_owned_pct >= $PCT$
}

# The mission is running (its homeland list is only filled while it runs).
mlhu_mission_running_trigger = {
	any_in_list = {
		variable = mlhu_homeland
		always = yes
	}
}

# One tree or the other (v1.13). Players: checked live. AI: the answer cached by
# mlhu_tree_cache_pulse (up to half a year old), so AI nations don't run ~500 task checks
# every time the game looks at their available missions.
mlhu_invictus_tree_blocks_trigger = {
	trigger_if = {
		limit = { is_ai = no }
		mlhu_invictus_tree_started_trigger = yes
	}
	trigger_else = {
		has_variable = mlhu_invictus_tree_started
	}
}
