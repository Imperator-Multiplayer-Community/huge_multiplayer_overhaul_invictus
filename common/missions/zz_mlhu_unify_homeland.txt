# Unify Your Home Lands (MLHU) - requested by Nabber.
# Generic tree for every nation: unite the territories of your capital's region where your people
# are the majority. "Your people" is the culture that is the majority in your capital; if that culture
# has no territory left to unite in the region, your primary culture is used instead.
# Limited to the capital's region so nations whose culture spread far (Macedonians from Greece to
# India, Punic from Iberia to Libya) don't get a homeland spanning half the map.
# The homeland is fixed when the mission starts (variable list mlhu_homeland on the country).
# v1.12: one tree or the other - hidden once the nation has started an Invictus mission tree.
# Progress is measured by a monthly event (mlhu.1) instead of list triggers inside the tasks, and
# every reward is given by an event (mlhu.11-17). The first four tasks take a year each.

mlhu_unify_homeland = {
	header = "mission_image_general"
	icon = "general_1"

	repeatable = no

	# Weight 1, same as Invictus's generic conquest mission, so historical trees
	# (Rome 3, Sparta 5, Carthage 10/50, ...) are preferred when available.
	chance = {
		factor = 1
	}

	ai_chance = {
		factor = 1
	}

	potential = {
		NOT = { has_variable = mlhu_cooldown }
		NOT = { has_variable = mlhu_homeland_done }
		is_subject = no
		has_civil_war = no
		# No invictus_mission_block_trigger here on purpose: after the one-province split many
		# nations it blocks (Hellenic, Macedonian Diadochi fragments...) can no longer start their
		# historical trees. Historical trees still win for the AI through the chance weights.
		OR = {
			mlhu_capital_kin_nearby_trigger = yes
			mlhu_primary_kin_nearby_trigger = yes
		}
		# One tree or the other: hidden once the nation has started an Invictus mission tree.
		mlhu_invictus_tree_started_trigger = no
	}

	abort = {
		ai_mission_back_out_trigger = yes
	}

	on_start = {
		start_mission_ai_effect = yes
		clear_variable_list = mlhu_homeland
		if = {
			# The people of our capital still have kin to unite in our region
			limit = { mlhu_capital_kin_nearby_trigger = yes }
			capital_scope.region = {
				every_region_province = {
					limit = {
						is_sea = no
						has_owner = yes
						dominant_province_culture = root.capital_scope.culture
					}
					root = {
						add_to_variable_list = { name = mlhu_homeland target = prev }
					}
				}
			}
		}
		else = {
			# Fallback: our primary culture's kin in our region
			capital_scope.region = {
				every_region_province = {
					limit = {
						is_sea = no
						has_owner = yes
						dominant_province_culture = root.culture
					}
					root = {
						add_to_variable_list = { name = mlhu_homeland target = prev }
					}
				}
			}
		}
		hidden_effect = {
			mlhu_update_progress_effect = yes	# first reading; then monthly by event mlhu.1
		}
	}

	on_abort = {
		custom_tooltip = mlhu_cooldown_tt
		hidden_effect = {
			mlhu_clear_progress_effect = yes
		}
		set_variable = {
			name = mlhu_cooldown
			days = 7300
		}
	}

	on_completion = {
		complete_mission_effect = yes
		hidden_effect = {
			mlhu_clear_progress_effect = yes
		}
	}

	# 1. Call of Kinship (1 year) - claim every homeland territory we don't hold
	mlhu_task_1 = {
		icon = task_political
		duration = 365

		allow = {
			political_influence >= 40
		}

		on_start = {
			add_political_influence = -40
		}

		on_completion = {
			show_as_tooltip = {
				custom_tooltip = mlhu_task_1_tt
			}
			trigger_event = { id = mlhu.11 }
		}
	}

	# 2. Secure the Heartland (1 year) - own a quarter of the homeland
	mlhu_task_2 = {
		icon = "task_battle"
		requires = { mlhu_task_1 }
		duration = 365

		highlight = {
			scope:province = {
				NOT = { owner = root }
			}
			root = { is_target_in_variable_list = { name = mlhu_homeland target = scope:province } }
		}

		allow = {
			custom_tooltip = {
				text = mlhu_task_2_req_tt
				mlhu_owns_homeland_share_trigger = { PCT = 25 }
			}
		}

		on_completion = {
			show_as_tooltip = {
				add_stability = 5
				add_country_modifier = {
					name = mlhu_rallied_kin
					duration = 3650
				}
			}
			trigger_event = { id = mlhu.12 }
		}
	}

	# 3. Bonds of Blood (1 year) - any nation as ally or subject (kin no longer required: after
	# conquering our kin there may be none left, and every nation should be able to finish the tree)
	mlhu_task_3 = {
		icon = task_diplomatic
		requires = { mlhu_task_1 }
		duration = 365

		allow = {
			custom_tooltip = {
				text = mlhu_task_3_req_tt
				any_country = {
					NOT = { this = root }
					has_land = yes
					OR = {
						is_subject_of = root
						alliance_with = root
					}
				}
			}
		}

		on_completion = {
			show_as_tooltip = {
				add_country_modifier = {
					name = mlhu_bonds_of_blood
					duration = 3650
				}
			}
			trigger_event = { id = mlhu.13 }
		}
	}

	# 4. Heart of Our People (1 year) - a populous capital
	mlhu_task_4 = {
		icon = "task_economical"
		requires = { mlhu_task_1 }
		duration = 365

		allow = {
			capital_scope = {
				total_population >= 20
			}
		}

		on_completion = {
			show_as_tooltip = {
				capital_scope = {
					add_permanent_province_modifier = {
						name = mlhu_heart_of_the_people
						duration = -1
					}
				}
			}
			trigger_event = { id = mlhu.14 }
		}
	}

	# 5. Half the Homeland - own half, renew claims
	mlhu_task_5 = {
		icon = "task_battle"
		requires = { mlhu_task_2 }

		highlight = {
			scope:province = {
				NOT = { owner = root }
			}
			root = { is_target_in_variable_list = { name = mlhu_homeland target = scope:province } }
		}

		allow = {
			custom_tooltip = {
				text = mlhu_task_5_req_tt
				mlhu_owns_homeland_share_trigger = { PCT = 50 }
			}
		}

		on_completion = {
			show_as_tooltip = {
				add_stability = 5
				add_political_influence = 50
				custom_tooltip = mlhu_task_1_tt
			}
			trigger_event = { id = mlhu.15 }
		}
	}

	# 6. Kin Under One Roof - own three quarters
	mlhu_task_6 = {
		icon = "task_battle"
		requires = { mlhu_task_5 mlhu_task_3 }

		highlight = {
			scope:province = {
				NOT = { owner = root }
			}
			root = { is_target_in_variable_list = { name = mlhu_homeland target = scope:province } }
		}

		allow = {
			custom_tooltip = {
				text = mlhu_task_6_req_tt
				mlhu_owns_homeland_share_trigger = { PCT = 75 }
			}
		}

		on_completion = {
			show_as_tooltip = {
				add_country_modifier = {
					name = mlhu_kin_under_one_roof
					duration = 3650
				}
			}
			trigger_event = { id = mlhu.16 }
		}
	}

	# 7. The Homeland United - own 90% (final)
	mlhu_task_7 = {
		icon = task_heracles
		final = yes
		requires = { mlhu_task_6 mlhu_task_4 }

		highlight = {
			scope:province = {
				NOT = { owner = root }
			}
			root = { is_target_in_variable_list = { name = mlhu_homeland target = scope:province } }
		}

		allow = {
			stability >= 40
			custom_tooltip = {
				text = mlhu_task_7_req_tt
				mlhu_owns_homeland_share_trigger = { PCT = 90 }
			}
		}

		on_completion = {
			show_as_tooltip = {
				add_stability = 10
				add_country_modifier = {
					name = mlhu_homeland_united
					duration = -1
				}
			}
			hidden_effect = {
				set_variable = mlhu_homeland_done
			}
			trigger_event = { id = mlhu.17 }
		}
	}
}
