# Foreign-tradition malus: keeps one -75% counter-modifier on every owned
# tradition that does not belong to the country's primary-culture tree.
# There is no engine hook for "primary culture changed" or "tradition picked",
# so a refresh catches both (and fixes anything else that drifts): monthly for
# players, once a year for AI nations (v1.13 - ~1,500 AI nations made the monthly
# refresh the biggest flat script cost in the mod).

monthly_country_pulse = {
	on_actions = {
		mlft_tradition_malus_pulse
		mlft_tradition_malus_ai_pulse
		mlft_food_share_pulse
		mllx_levy_xp_pulse
		mlat_auto_trade_pulse
		mlhu_progress_pulse
		mlhu_tree_cache_pulse
	}
}

mlft_tradition_malus_pulse = {
	trigger = {
		is_ai = no
	}
	effect = {
		mlft_refresh_tradition_malus_effect = yes
	}
}

# AI nations: first month of the game, then once a year (each nation on its own date).
mlft_tradition_malus_ai_pulse = {
	trigger = {
		is_ai = yes
		NOT = { has_variable = mlft_ai_malus_cd }
	}
	effect = {
		mlft_refresh_tradition_malus_effect = yes
		set_variable = {
			name = mlft_ai_malus_cd
			days = 365
		}
	}
}


# Open granaries for allies (ported from HMO: Huge Multiplayer Overhaul).
mlft_food_share_pulse = {
	events = {
		mlft_food_share.1
	}
}


# Levy XP (ported from HMO): record levy starting XP and keep levy XP decay
# near zero while no legion law is enacted.
mllx_levy_xp_pulse = {
	effect = {
		mllx_refresh_levy_xp_retention_effect = yes
	}
	events = {
		mllx_levy_xp.1
	}
}


# Auto-trade helper: fills free trade routes of player states that have automated trading on.
mlat_auto_trade_pulse = {
	trigger = {
		is_ai = no
	}
	effect = {
		mlat_auto_trade_effect = yes
	}
}


# Unify Your Home Lands: measure homeland ownership once a month while the mission runs.
mlhu_progress_pulse = {
	events = {
		mlhu.1
	}
}

# Unify Your Home Lands, one tree or the other: AI nations check the ~500 Invictus first
# tasks twice a year and remember the answer, instead of re-checking them every time the
# game looks at which missions they can start (players are still checked live).
mlhu_tree_cache_pulse = {
	trigger = {
		is_ai = yes
		NOT = { has_variable = mlhu_invictus_tree_started }
		NOT = { has_variable = mlhu_homeland_done }
		NOT = { has_variable = mlhu_tree_check_cd }
	}
	effect = {
		mlhu_refresh_tree_cache_effect = yes
	}
}


# Truces: lengthen the truces of a war that just ended (by game date and nation rank).
on_ending_war = {
	on_actions = {
		mltr_truce_on_war_end
	}
}

mltr_truce_on_war_end = {
	effect = {
		mltr_extend_truces_effect = yes
	}
}
